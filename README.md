from tkinter import \*
from tkinter import messagebox
import sqlite3
con=sqlite3.connect("databaseconnect.db")
con.execute("create table if not exists details(name char(40),rollno char(40),email char(40),phonenumber char(40),teamname char(40),teamidea char(100),teamno char(30))")
win=Tk()
win.geometry("700x700")
name=StringVar()
rollno=StringVar()
email=StringVar()
phonenumber=StringVar()
teamname=StringVar()
teamno=StringVar()
teamidea=StringVar()
def reset():
    rollno.set("")
    name.set("")
    email.set("")
    phonenumber.set("")
    teamname.set("")
    teamno.set("")
    teamidea.set("")
def save():
    rollno1=rollno.get()
    name1=name.get()
    email1=email.get()
    phone1=phonenumber.get()
    teamname1=teamname.get()
    teamno1=teamno.get()
    teamidea1=teamidea.get()
    con.execute("insert into details values (?,?,?,?,?,?,?)",(rollno1,name1,email1,phone1,teamname1,teamno1,teamidea1))
    con.commit()
    messagebox.showinfo("event registration page","information saved")
def update():
    rollno1=rollno.get()
    name1=name.get()
    email1=email.get()
    phone1=phonenumber.get()
    teamname1=teamname.get()
    teamno1=teamno.get()
    teamidea1=teamidea.get()
    con.execute("update details set name=?,email=?,phonenumber=?,teamname=?,teamno=?,teamidea=? where rollno=?",(name1,email1,phone1,teamname1,teamno1,teamidea1,rollno1))
    con.commit()
    messagebox.showinfo("event registritration page","update confirmed")
def showall():
    show = con.execute("select \* from details")
    for row in show:
        print(row[0],"",row[1],"",row[2],"",row[3],"",row[4],"",row[5],"",row[6])
win.title("Event Registration Page")
l1=Label(win,text="Event Registration Page",font=('arial',20,'bold')).grid(columnspan=4)
l1=Label(win,text="Fill the following details",font=('bodoni',15)).grid(columnspan=4)
l2=Label(win,text="Candidate name",font=('bodoni',10)).grid(row=3,column=0)
e1=Entry(win,textvariable=name).grid(row=3,column=1)
l3=Label(win,text="Roll number",font=('bodoni',10)).grid(row=4,column=0)
e3=Entry(win,textvariable=rollno).grid(row=4,column=1)
l4=Label(win,text="Email",font=('bodoni',10)).grid(row=5,column=0)
e4=Entry(win,textvariable=email).grid(row=5,column=1)
l5=Label(win,text="phone number",font=('bodoni',10)).grid(row=6,column=0)
e5=Entry(win,textvariable=phonenumber).grid(row=6,column=1)
l6=Label(win,text="team name",font=('bodoni',10)).grid(row=7,column=0)
e6=Entry(win,textvariable=teamname).grid(row=7,column=1)
l7=Label(win,text="number of member in a team",font=('bodoni',10)).grid(row=8,column=0)
e7=Entry(win,textvariable=teamno).grid(row=8,column=1)
l8=Label(win,text="team idea",font=('bodoni',15)).grid(row=9,column=0)
e8=Entry(win,textvariable=teamidea).grid(row=9,column=1)
B1=Button(win,text="reset",font=('arial',15,'bold'),bd=8,command=reset).grid(row=10,column=0)
b2=Button(win,text="update",font=('arial',15,'bold'),bd=8,command=update).grid(row=10,column=1)
b3=Button(win,text="save",font=('arial',15,'bold'),bd=8,command=save).grid(row=10,column=3)
b4=Button(win,text="see my details",font=('arial',15,'bold'),bd=8,command=showall).grid(row=11,column=1)
win.mainloop
